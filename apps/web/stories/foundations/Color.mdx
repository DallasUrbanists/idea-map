export const swatchForeground = (hex) => {
  const luminance = (value) => {
    const channels = value
      .match(/[0-9a-f]{2}/gi)
      .map((channel) => parseInt(channel, 16) / 255);
    const linear = channels.map((channel) =>
      channel <= 0.04045 ? channel / 12.92 : ((channel + 0.055) / 1.055) ** 2.4,
    );
    return 0.2126 * linear[0] + 0.7152 * linear[1] + 0.0722 * linear[2];
  };
  const contrast = (a, b) => (Math.max(a, b) + 0.05) / (Math.min(a, b) + 0.05);
  const background = luminance(hex);
  const foregrounds = [
    "--color-swatch-ink",
    "--color-swatch-light",
    "--color-swatch-black",
  ];
  return foregrounds.find(
    (token) => contrast(background, luminance(getColorValue(token))) >= 4.5,
  );
};

export const ColorSwatch = ({ token, theme, fixed = false }) => {
  const hex = getColorValue(token, theme);
  const foreground = swatchForeground(hex);
  return (
    <td style={{ backgroundColor: fixed ? `var(${token})` : hex, color: `var(${foreground})` }}>
      {hex}
    </td>
  );
};

export const ColorRows = ({ rows }) =>
  rows.map(({ token }) => (
    <tr key={token}>
      <th scope="row"><code>{token}</code></th>
      <ColorSwatch token={token} theme="light" />
      <ColorSwatch token={token} theme="dark" />
    </tr>
  ));

export const SemanticColorRows = ({ tokens }) => tokens.map((token) => (
  <tr key={token}>
    <th scope="row"><code>{token}</code></th>
    {["light", "dark"].map((theme) => {
      const palette = getPaletteToken(token, theme);
      const foreground = swatchForeground(getColorValue(palette));
      return <td key={theme} style={{ backgroundColor: `var(${palette})`, color: `var(${foreground})` }}><code>{palette}</code></td>;
    })}
  </tr>
));

# Color

Common Ground draws on neighborhood streets, buildings, trees, and public art.
Neutral surfaces leave room for content, blue identifies primary actions, and
supporting colors add warmth. This reference follows the section structure of
[Signal's color guide](https://signal.chitransit.app/foundations/color) and
documents Idea Map's current tokens.

actions and lilac for selection; dark mode pairs slate surfaces with lighter
blue actions. Coral, canopy, and marigold add small moments of color.

Use Asphalt text on the supporting coral, traffic cone, canopy, marigold, and lilac fills.
Keep most of a page neutral, make blue the main action color, and choose
supporting accents sparingly. The palette lists available colors rather than
the proportions of a finished screen.

Use white text on the base transit blue and Asphalt text on coral, traffic cone,
canopy, marigold, and lilac. White text on those supporting colors does not
meet the 4.5:1 contrast requirement for ordinary text.

Check ordinary text at 4.5:1 or higher and essential control boundaries and
focus rings at 3:1 or higher against adjacent colors, including hover and
selection states. Use a visible label, mark, or icon alongside color for status.

## DART public transit colors

DART identifies its rail services by name and color: Red, Blue, Green, Orange,
and Silver. Keep the line name available in text or an accessible label rather
than relying on color alone.

The following are reference colors reported for line icons from DART's
[2025 rail guide](https://dartorgcmsblob.dart.org/prod/docs/default-source/silverline/dart-silverline-guide_oct-2025.pdf).
The bus reference colors come from the category backgrounds on DART's
[bus service page](https://www.dart.org/guide/transit-and-use/bus-routes).
These tokens do not establish a verified contrast treatment for map lines.
Use a text label or another distinguishing mark alongside each color, and
check contrast against its actual background.
